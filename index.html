/* ============================================================ إذاعة نبض الإخبارية — الواجهة البرمجية (Google Apps Script) تربط صفحة index.html بملف جوجل شيت. خطوات الاستخدام: 1) افتح ملف جوجل شيت ← Extensions ← Apps Script 2) احذف كل الكود الموجود والصق هذا الملف بالكامل 3) غيّر SHEET_NAME بالأسفل إذا كان اسم الورقة مختلفاً 4) شغّل الدالة setup مرة واحدة (اختر setup من القائمة ثم Run) 5) Deploy ← New deployment ← Web app Execute as : Me Who has access : Anyone 6) انسخ الرابط المنتهي بـ /exec وضعه في API_URL داخل index.html ============================================================ */ var SHEET_NAME = 'الورقة1'; // اسم الورقة داخل ملف جوجل شيت var DEFAULT_PASS = '1234'; // الرقم السري الأولي للوحة التحكم var HEADERS = ['id', 'date', 'title', 'img', 'content']; /* ---------- أدوات مساعدة ---------- */ function getSheet_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(SHEET_NAME); if (!sh) sh = ss.insertSheet(SHEET_NAME); if (sh.getLastRow() === 0) { sh.appendRow(HEADERS); } return sh; } function getPass_() { var p = PropertiesService.getScriptProperties().getProperty('ADMIN_PASS'); return p || DEFAULT_PASS; } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /* ---------- تجهيز أولي: شغّلها مرة واحدة ---------- */ function setup() { var sh = getSheet_(); sh.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); sh.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold'); sh.setFrozenRows(1); PropertiesService.getScriptProperties().setProperty('ADMIN_PASS', DEFAULT_PASS); Logger.log('تم التجهيز بنجاح. الرقم السري الحالي: ' + DEFAULT_PASS); } /* ---------- قراءة الأخبار ---------- */ function doGet(e) { var sh = getSheet_(); var values = sh.getDataRange().getValues(); var headers = values.shift(); var data = values .filter(function (row) { return row[0] !== '' && row[0] !== null; }) .map(function (row) { var obj = {}; headers.forEach(function (h, i) { obj[h] = row[i]; }); obj.id = Number(obj.id); return obj; }) .sort(function (a, b) { return b.id - a.id; }); // الأحدث أولاً return json_(data); } /* ---------- الإضافة والحذف وتغيير الرقم السري ---------- */ function doPost(e) { var body; try { body = JSON.parse(e.postData.contents); } catch (err) { return json_({ ok: false, msg: 'bad request' }); } var action = body.action; // كل العمليات تتطلب الرقم السري الصحيح if (body.pass !== getPass_()) { return json_({ ok: false, msg: 'unauthorized' }); } var sh = getSheet_(); if (action === 'auth') { return json_({ ok: true }); } if (action === 'add') { var a = body.article || {}; var id = Date.now(); sh.appendRow([id, a.date || '', a.title || '', a.img || '', a.content || '']); return json_({ ok: true, id: id }); } if (action === 'delete') { var target = Number(body.id); var ids = sh.getRange(2, 1, Math.max(sh.getLastRow() - 1, 1), 1).getValues(); for (var i = 0; i < ids.length; i++) { if (Number(ids[i][0]) === target) { sh.deleteRow(i + 2); return json_({ ok: true }); } } return json_({ ok: false, msg: 'not found' }); } if (action === 'clearAll') { if (sh.getLastRow() > 1) { sh.deleteRows(2, sh.getLastRow() - 1); } return json_({ ok: true }); } if (action === 'changePass') { var np = String(body.newPass || '').trim(); if (!np) return json_({ ok: false, msg: 'empty' }); PropertiesService.getScriptProperties().setProperty('ADMIN_PASS', np); return json_({ ok: true }); } return json_({ ok: false, msg: 'unknown action' }); }
